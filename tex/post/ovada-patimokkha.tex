
\section{Ovāda-pātimokkha-gāthā}
\label{ovada-patimokkha-gatha}

\begin{intro}
	\anglebracketleft\ \hspace{-0.5mm}Handa mayaṁ ovāda-pātimokkha gāthāyo bhaṇāmase \hspace{-0.5mm}\anglebracketright\
\end{intro}

%% \medskip

\begin{hanging-indent}
Uddiṭṭhaṁ kho tena bhagavatā jānatā passatā arahatā sammā-sambuddhena
\end{hanging-indent}
Ovāda-pātimokkhaṁ tīhi gāthāhi
\begin{hanging-indent}
Khantī paramaṁ tapo tītikkhā\\
Nibbānaṁ paramaṁ vadanti buddhā
\end{hanging-indent}

\begin{hanging-indent}
Na hi pabbajito par'ūpaghātī\\
Samaṇo hoti paraṁ viheṭhayanto
\end{hanging-indent}

\begin{hanging-indent}
Sabba-pāpassa akaraṇaṁ\\
Kusalass'ūpasampadā
\end{hanging-indent}

\begin{hanging-indent}
Sacitta-pariyodapanaṁ\\
Etaṁ buddhāna'sāsanaṁ
\end{hanging-indent}

\begin{hanging-indent}
Anūpavādo anūpaghāto\\
Pātimokkhe ca saṁvaro
\end{hanging-indent}

\begin{hanging-indent}
Mattaññutā ca bhattasmiṁ\\
Pantañ'ca sayan'āsanaṁ
\end{hanging-indent}

\begin{hanging-indent}
Adhicitte ca āyogo\\
Etaṁ buddhāna'sāsanan'ti
\end{hanging-indent}

\suttaRef{[Dhp 183-185]}

\clearpage



\section{The Pātimokkha Exhortation}
\label{ovada-patimokkha-gatha-english}

\begin{intro}
	\anglebracketleft\ \hspace{-0.5mm}Now let us recite the verses of the pātimokkha exhortation \hspace{-0.5mm}\anglebracketright\
\end{intro}

\begin{hanging-indent}
This was summarized by the Blessed One, the One who Knows, the One who Sees, the Worthy One, the Perfectly Enlightened One
\end{hanging-indent}
the Pātimokkha Exhortation with three verses:
\begin{hanging-indent}
Patient endurance is the highest austerity\\
The Buddhas say Nibbāna is supreme
\end{hanging-indent}

\begin{hanging-indent}
Not a renunciant is one who injures others\\
Whoever troubles others can't be called a monk
\end{hanging-indent}

\begin{hanging-indent}
Not doing any evil\\
To be committed to the good
\end{hanging-indent}

\begin{hanging-indent}
To purify one's mind\\
These are the teachings of all Buddhas
\end{hanging-indent}

\begin{hanging-indent}
Not to insult and not to injure\\
To live restrained by training rules
\end{hanging-indent}

\begin{hanging-indent}
Knowing one's measure at the meal\\
Retreating to a lonely place
\end{hanging-indent}

\begin{hanging-indent}
Devotion to the higher mind\\
These are the teachings of all Buddhas
\end{hanging-indent}

\suttaRef{[Dhp 183-185]}

\clearpage
